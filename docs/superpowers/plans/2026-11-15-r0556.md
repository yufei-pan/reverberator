/etc/fstab once per look, an unreadable one being a problem.

	'setgid': 'before_chmod',

	'own_xattr': 'copy_xattrs',

# step 1's three places: V0's walk and settle() of a folder a batch made (mkdir C), where C is new and inherits from P what the
# case names; and a batch's attrib of C, where C is right in the vault and the batch gives the monitored C what the case names
# (chmod g+s, a default ACL, user.note), which settle() copies too
PLACES = ['first_version', 'settle', 'attrib']

@pytest.mark.parametrize('case', CASES)
@pytest.mark.parametrize('place', PLACES)
